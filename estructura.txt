name: Dev Diary

on:
  schedule:
    - cron: '0 23 * * *'   # 23:00 UTC = 6 PM Cuba
  workflow_dispatch:

jobs:
  daily-diary:
    runs-on: ubuntu-latest
    permissions:
      contents: read
      issues: read

    steps:
      - name: Recopilar actividad del día
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          DISCORD_WEBHOOK: ${{ secrets.DISCORD_WEBHOOK_URL }}
          OWNER: ${{ github.repository_owner }}
        run: |
          set -e

          # Repositorios a revisar (edita esta lista si quieres)
          REPOS=(
            "Traeume-des-Blauen"
            "nexiria"
            "yozora-store-android"
            "Dibujama_Android"
            "dibujama"
            "converty"
            "baby-medina-android"
          )

          TODAY=$(date -u +"%Y-%m-%d")
          SINCE=$(date -u -d "24 hours ago" +"%Y-%m-%dT%H:%M:%SZ")

          TOTAL_COMMITS=0
          REPOS_ACTIVOS=""
          RELEASES_HOY=0
          ISSUES_NUEVOS=0

          for REPO in "${REPOS[@]}"; do
            # Commits de las últimas 24h
            COMMITS=$(curl -s -H "Authorization: Bearer $GH_TOKEN" \
              "https://api.github.com/repos/$OWNER/$REPO/commits?since=$SINCE&per_page=100" \
              | jq 'length' 2>/dev/null || echo "0")

            if [ "$COMMITS" -gt 0 ] 2>/dev/null; then
              TOTAL_COMMITS=$((TOTAL_COMMITS + COMMITS))
              REPOS_ACTIVOS="${REPOS_ACTIVOS}\n• \`${REPO}\`: ${COMMITS} commit(s)"
            fi

            # Releases publicados hoy
            RELEASES=$(curl -s -H "Authorization: Bearer $GH_TOKEN" \
              "https://api.github.com/repos/$OWNER/$REPO/releases?per_page=10" \
              | jq --arg since "$SINCE" '[.[] | select(.published_at > $since)] | length' 2>/dev/null || echo "0")
            RELEASES_HOY=$((RELEASES_HOY + RELEASES))

            # Issues abiertos hoy
            ISSUES=$(curl -s -H "Authorization: Bearer $GH_TOKEN" \
              "https://api.github.com/repos/$OWNER/$REPO/issues?state=open&since=$SINCE&per_page=50" \
              | jq --arg since "$SINCE" '[.[] | select(.created_at > $since)] | length' 2>/dev/null || echo "0")
            ISSUES_NUEVOS=$((ISSUES_NUEVOS + ISSUES))
          done

          # Construir mensaje
          if [ "$TOTAL_COMMITS" -gt 0 ] || [ "$RELEASES_HOY" -gt 0 ] || [ "$ISSUES_NUEVOS" -gt 0 ]; then
            MSG="📓 **Diario del desarrollador — ${TODAY}**\n"
            MSG="${MSG}• **${TOTAL_COMMITS}** commit(s) en total"
            if [ -n "$REPOS_ACTIVOS" ]; then
              MSG="${MSG}\n• Repos con actividad:${REPOS_ACTIVOS}"
            fi
            if [ "$RELEASES_HOY" -gt 0 ]; then
              MSG="${MSG}\n• **${RELEASES_HOY}** release(s) publicado(s)"
            fi
            if [ "$ISSUES_NUEVOS" -gt 0 ]; then
              MSG="${MSG}\n• **${ISSUES_NUEVOS}** issue(s) nuevo(s)"
            fi
            MSG="${MSG}\n\nMañana más. 💙"
          else
            COMENTARIOS=(
              "Hoy no hubo movimiento. Supongo que descansaste. Bien por ti. 💙"
              "Día tranquilo. A veces hace falta. 💙"
              "Nada por aquí. El silencio también cuenta. 💙"
              "Sin actividad hoy. Mañana será otro día. 💙"
            )
            RANDOM_INDEX=$((RANDOM % ${#COMENTARIOS[@]}))
            MSG="📓 **Diario del desarrollador — ${TODAY}**\n\n${COMENTARIOS[$RANDOM_INDEX]}"
          fi

          PAYLOAD=$(jq -n --arg content "$MSG" '{content: $content}')
          curl -s -X POST -H "Content-Type: application/json" -d "$PAYLOAD" "$DISCORD_WEBHOOK"
          echo "Diario enviado."

name: Download Notifier

on:
  schedule:
    - cron: '*/15 * * * *'
  workflow_dispatch:

jobs:
  check-downloads:
    runs-on: ubuntu-latest
    permissions:
      contents: read

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Consultar descargas y notificar a Discord
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          DISCORD_WEBHOOK: ${{ secrets.DISCORD_WEBHOOK_URL }}
          GIST_TOKEN: ${{ secrets.GIST_TOKEN }}
          GIST_ID: ${{ secrets.GIST_ID }}
          REPO: ${{ github.repository }}
        run: |
          set -e

          NOW_DATE=$(date -u +"%Y-%m-%d")
          NOW_HOUR=$(date -u +"%H")
          echo "Fecha actual: $NOW_DATE, hora: $NOW_HOUR"

          # 1. Obtener releases
          RELEASES=$(curl -s -H "Authorization: Bearer $GH_TOKEN" \
            "https://api.github.com/repos/$REPO/releases?per_page=100")

          # 2. Construir {app: count} sumando solo APKs, agrupado por app
          CURRENT=$(echo "$RELEASES" | jq -c '
            [.[] | . as $release |
              .assets[] |
              select(.name | endswith(".apk")) |
              {
                app: ($release.tag_name | sub("_v[0-9.]+$"; "")),
                count: .download_count
              }
            ] | group_by(.app) | map({
              key: .[0].app,
              value: (map(.count) | add)
            }) | from_entries
          ')

          echo "Conteo actual: $CURRENT"

          # 3. Leer Gist anterior
          GIST_CONTENT=$(curl -s -H "Authorization: Bearer $GIST_TOKEN" \
            "https://api.github.com/gists/$GIST_ID" \
            | jq -r '.files["downloads.json"].content // "{\"current\":{},\"daily\":{}}"')

          echo "Gist anterior: $GIST_CONTENT"

          PREVIOUS=$(echo "$GIST_CONTENT" | jq -c '.current // {}')
          DAILY=$(echo "$GIST_CONTENT" | jq -c '.daily // {}')

          # 4. Comparar y construir mensajes + deltas del día
          MESSAGES=""
          TODAY_DELTAS="{}"
          APPS=$(echo "$CURRENT" | jq -r 'keys[]')

          for APP in $APPS; do
            OLD=$(echo "$PREVIOUS" | jq -r --arg a "$APP" '.[$a] // 0')
            NEW=$(echo "$CURRENT"  | jq -r --arg a "$APP" '.[$a] // 0')
            DIFF=$((NEW - OLD))

            if [ "$DIFF" -gt 0 ]; then
              MESSAGES="${MESSAGES}${APP}: ${DIFF} nueva(s) descarga(s)\n"

              TODAY_DELTAS=$(echo "$TODAY_DELTAS" | jq -c \
                --arg a "$APP" --argjson d "$DIFF" \
                '.[$a] = ((.[$a] // 0) + $d)')
            fi
          done

          # 5. Enviar notificación de descargas nuevas (solo si hay)
          if [ -n "$MESSAGES" ]; then
            echo "Enviando notificación de descargas nuevas..."
            MSG="Nuevas descargas detectadas:\n${MESSAGES}"
            PAYLOAD=$(jq -n --arg content "$MSG" '{content: $content}')
            curl -s -X POST \
              -H "Content-Type: application/json" \
              -d "$PAYLOAD" \
              "$DISCORD_WEBHOOK"
            echo "Notificado."
          else
            echo "Sin novedades."
          fi

          # 6. Actualizar daily con los deltas de hoy
          UPDATED_DAILY=$(jq -n \
            --argjson daily "$DAILY" \
            --arg date "$NOW_DATE" \
            --argjson today "$TODAY_DELTAS" \
            '
            $daily as $d |
            ($d[$date] // {}) as $existing |
            $d + { ($date): (
              ($existing | keys) + ($today | keys) | unique |
              map({ key: ., value: (($existing[.] // 0) + ($today[.] // 0)) }) |
              from_entries
            )}
            ')

          # 7. Construir nuevo Gist
          NEW_GIST=$(jq -n \
            --argjson current "$CURRENT" \
            --argjson daily "$UPDATED_DAILY" \
            '{ current: $current, daily: $daily }')

          GIST_BODY=$(jq -n --arg content "$NEW_GIST" '{
            files: { "downloads.json": { content: $content } }
          }')
          curl -s -X PATCH \
            -H "Authorization: Bearer $GIST_TOKEN" \
            -H "Content-Type: application/json" \
            -d "$GIST_BODY" \
            "https://api.github.com/gists/$GIST_ID" > /dev/null

          echo "Gist actualizado."

          # 8. Resumen de medianoche (solo si es la hora 0)
          if [ "$NOW_HOUR" = "00" ]; then
            echo "Hora de resumen diario..."

            DAY_TOTALS=$(echo "$UPDATED_DAILY" | jq -c --arg d "$NOW_DATE" '.[$d] // {}')

            if [ "$DAY_TOTALS" != "{}" ] && [ "$(echo "$DAY_TOTALS" | jq 'length')" -gt 0 ]; then
              SUMMARY="Resumen del dia ${NOW_DATE}:\n"
              for APP in $(echo "$DAY_TOTALS" | jq -r 'keys[]'); do
                COUNT=$(echo "$DAY_TOTALS" | jq -r --arg a "$APP" '.[$a]')
                SUMMARY="${SUMMARY}${APP}: ${COUNT} descarga(s)\n"
              done

              PAYLOAD=$(jq -n --arg content "$SUMMARY" '{content: $content}')
              curl -s -X POST \
                -H "Content-Type: application/json" \
                -d "$PAYLOAD" \
                "$DISCORD_WEBHOOK"
              echo "Resumen diario enviado."
            else
              echo "Sin descargas hoy. No se envia resumen."
            fi
          fi

name: Issue Responder

on:
  issues:
    types: [opened]
  schedule:
    - cron: '30 1 * * *'
  workflow_dispatch:

jobs:
  responder:
    runs-on: ubuntu-latest
    permissions:
      issues: write

    steps:
      - name: Etiquetar y responder issue nuevo
        if: github.event_name == 'issues'
        uses: actions/github-script@v7
        with:
          script: |
            const issue = context.payload.issue;
            const title = issue.title.toLowerCase();
            const body = (issue.body || '').toLowerCase();
            const text = title + ' ' + body;

            let label = '';
            let message = '';

            if (text.match(/\b(bug|error|falla|fallo|no funciona|crash|se cierra)\b/)) {
              label = 'bug';
              message = `¡Qué sucede! 👋 Gracias por reportar.\n\nPara poder ayudarte bien, necesito algunos datos:\n• Modelo de teléfono\n• Versión de Android\n• Versión de la app\n• Qué pasos seguiste hasta que falló\n\nEn cuanto me lo digas, le echo un ojo. 💙`;
            } else if (text.match(/\b(sugerencia|idea|me gustaría|podría|sería bueno|feature|añadir)\b/)) {
              label = 'sugerencia';
              message = `¡Qué sucede! 👋 Gracias por la idea.\n\nCuéntame un poco más: ¿qué te gustaría ver y para qué lo usarías? Así puedo valorarlo mejor. 💙`;
            } else if (text.match(/\b(encargo|forja|presupuesto|contratar|personalizada|a medida)\b/)) {
              label = 'encargo';
              message = `¡Qué sucede! 👋 Gracias por el interés.\n\nPara encargar una app personalizada, necesito:\n• Descripción de lo que quieres\n• Presupuesto aproximado\n• Plazo que manejas\n\nLo mejor es que me escribas por WhatsApp y lo hablamos con calma. 💙`;
            } else {
              label = 'triage';
              message = `¡Qué sucede! 👋 Gracias por escribir.\n\nHe recibido tu issue y lo revisaré en cuanto pueda. Si me das algún detalle más, mejor. 💙`;
            }

            await github.rest.issues.addLabels({
              owner: context.repo.owner,
              repo: context.repo.repo,
              issue_number: issue.number,
              labels: [label]
            });

            await github.rest.issues.createComment({
              owner: context.repo.owner,
              repo: context.repo.repo,
              issue_number: issue.number,
              body: message
            });

            console.log(`Issue #${issue.number} etiquetado como "${label}" y respondido.`);

      - name: Notificar a Discord sobre issue nuevo
        if: github.event_name == 'issues'
        env:
          DISCORD_WEBHOOK: ${{ secrets.DISCORD_WEBHOOK_URL }}
        run: |
          ISSUE_NUM="${{ github.event.issue.number }}"
          ISSUE_TITLE="${{ github.event.issue.title }}"
          ISSUE_AUTHOR="${{ github.event.issue.user.login }}"
          ISSUE_URL="${{ github.event.issue.html_url }}"
          ISSUE_BODY="${{ github.event.issue.body }}"

          # Acortar el cuerpo si es muy largo
          BODY_PREVIEW=$(echo "$ISSUE_BODY" | head -c 200)
          if [ ${#ISSUE_BODY} -gt 200 ]; then
            BODY_PREVIEW="${BODY_PREVIEW}..."
          fi

          MSG="🐛 **Nuevo issue #${ISSUE_NUM}**\n"
          MSG="${MSG}• **Título:** ${ISSUE_TITLE}\n"
          MSG="${MSG}• **Autor:** @${ISSUE_AUTHOR}\n"
          if [ -n "$ISSUE_BODY" ]; then
            MSG="${MSG}• **Descripción:** ${BODY_PREVIEW}\n"
          fi
          MSG="${MSG}\n[Ver issue](${ISSUE_URL})"

          PAYLOAD=$(jq -n --arg content "$MSG" '{content: $content}')
          curl -s -X POST -H "Content-Type: application/json" -d "$PAYLOAD" "$DISCORD_WEBHOOK"
          echo "Notificado a Discord."

      - name: Cerrar issues inactivos
        if: github.event_name == 'schedule'
        uses: actions/stale@v9
        with:
          days-before-issue-stale: 30
          days-before-issue-close: 14
          stale-issue-label: 'stale'
          stale-issue-message: 'Oye, este issue lleva mucho sin moverse. Si ya no es relevante, lo cierro en 14 días. Si sigue vivo, avísame. 💙'
          close-issue-message: 'Lo cierro por inactividad. Si vuelve a ser relevante, reábrelo o crea uno nuevo. 💙'
          days-before-pr-stale: -1
          days-before-pr-close: -1
          repo-token: ${{ secrets.GITHUB_TOKEN }}

name: Release Apps and Build Catalog

on:
  push:
    branches: [main]
    paths:
      - 'apps/**'
  workflow_dispatch:

jobs:
  release-apps:
    name: Create/Update GitHub Releases
    runs-on: windows-latest
    permissions:
      contents: write

    outputs:
      changed_apps: ${{ steps.detect.outputs.changed_apps }}

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4
        with:
          fetch-depth: 2

      - name: Install powershell-yaml
        shell: pwsh
        run: Install-Module -Name powershell-yaml -Scope CurrentUser -Force

      - name: Detect changed apps
        id: detect
        shell: pwsh
        run: |
          # Obtener archivos cambiados en el último commit
          $changed = git diff --name-only HEAD~1 HEAD
          $appIds = @()
          foreach ($file in $changed) {
            if ($file -match '^apps/([^/]+)/') {
              $appIds += $Matches[1]
            }
          }
          $appIds = $appIds | Sort-Object -Unique
          Write-Host "Apps cambiadas: $($appIds -join ', ')"
          $json = $appIds | ConvertTo-Json -Compress
          if ($appIds.Count -eq 0) { $json = '[]' }
          "changed_apps=$json" | Out-File -FilePath $env:GITHUB_OUTPUT -Append -Encoding utf8

      - name: Create releases for changed apps
        if: steps.detect.outputs.changed_apps != '[]'
        shell: pwsh
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: |
          $appIds = '${{ steps.detect.outputs.changed_apps }}' | ConvertFrom-Json
          foreach ($appId in $appIds) {
            Write-Host "`n=== Procesando $appId ===" -ForegroundColor Cyan

            $appDir = "apps/$appId"
            $infoPath = "$appDir/info.md"
            if (-not (Test-Path $infoPath)) {
              Write-Warning "No hay info.md en $appDir"
              continue
            }

            # Leer frontmatter para obtener la version
            $bytes = [System.IO.File]::ReadAllBytes($infoPath)
            $content = [System.Text.Encoding]::UTF8.GetString($bytes)
            if ($content -notmatch '(?s)^\s*---\s*\r?\n(.*?)\r?\n---\s*\r?\n(.*)$') {
              Write-Warning "Frontmatter invalido en $infoPath"
              continue
            }
            $meta = ConvertFrom-Yaml $Matches[1]
            $version = $meta.version
            $appName = $meta.name

            $tag = "${appId}_v$version"
            Write-Host "Tag: $tag"

            # Buscar APK
            $apk = Get-ChildItem -Path $appDir -Filter "*.apk" | Select-Object -First 1
            if (-not $apk) {
              Write-Warning "No hay APK en $appDir"
              continue
            }

            # Ver si el release ya existe
            $existing = gh release view $tag 2>$null
            if ($LASTEXITCODE -eq 0) {
              Write-Host "Release $tag ya existe. Eliminando para recrear..." -ForegroundColor Yellow
              gh release delete $tag --yes --cleanup-tag
            }

            # Crear release y subir assets
            Write-Host "Creando release $tag..."
            $releaseArgs = @(
              "release", "create", $tag,
              "--title", "$appName $version",
              "--notes", "Release automatico de $appName version $version",
              "--latest"
            )

            # Assets: APK + icono si existe
            $assets = @($apk.FullName)
            $iconPath = "$appDir/icon.png"
            if (Test-Path $iconPath) {
              $assets += (Resolve-Path $iconPath).Path
            }

            gh @releaseArgs @assets

            if ($LASTEXITCODE -ne 0) {
              throw "Fallo al crear el release $tag"
            }

            Write-Host "Release $tag creado con $($assets.Count) assets." -ForegroundColor Green
          }

  build-catalog:
    name: Regenerate catalog.json
    needs: release-apps
    runs-on: windows-latest
    permissions:
      contents: write

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4
        with:
          ref: main

      - name: Install powershell-yaml
        shell: pwsh
        run: Install-Module -Name powershell-yaml -Scope CurrentUser -Force

      - name: Generate catalog.json
        shell: pwsh
        run: ./scripts/Generate-Catalog.ps1

      - name: Commit catalog.json if changed
        shell: pwsh
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "github-actions[bot]@users.noreply.github.com"
          git add docs/catalog.json -f
          if (git diff --cached --quiet) {
            Write-Host "Sin cambios en catalog.json"
          } else {
            git commit -m "chore: regenerate catalog.json [skip ci]"
            git push
          }

name: Repo Thermometer

on:
  schedule:
    - cron: '0 22 * * 0'
  workflow_dispatch:

jobs:
  thermometer:
    runs-on: ubuntu-latest
    permissions:
      contents: read
      issues: read

    steps:

name: Version Checker

on:
  push:
    branches: [main]
  schedule:
    - cron: '0 6 * * *'   # Una vez al día a las 6 AM UTC
  workflow_dispatch:

jobs:
  check-versions:
    runs-on: ubuntu-latest
    permissions:
      contents: read

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Comparar versiones con releases
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          DISCORD_WEBHOOK: ${{ secrets.DISCORD_WEBHOOK_URL }}
          REPO: ${{ github.repository }}
        run: |
          set -e

          MESSAGES=""

          for APP_DIR in apps/*/; do
            APP_ID=$(basename "$APP_DIR")
            INFO_PATH="${APP_DIR}info.md"

            if [ ! -f "$INFO_PATH" ]; then
              continue
            fi

            # Extraer versión del frontmatter
            VERSION=$(grep -m1 '^version:' "$INFO_PATH" | sed 's/version:\s*//' | tr -d '"' | tr -d "'" | xargs)

            if [ -z "$VERSION" ]; then
              MESSAGES="${MESSAGES}⚠️ **${APP_ID}**: no se pudo leer la versión del info.md\n"
              continue
            fi

            # Buscar último release de esta app
            TAG="${APP_ID}_v${VERSION}"
            RELEASE_EXISTS=$(curl -s -o /dev/null -w "%{http_code}" \
              -H "Authorization: Bearer $GH_TOKEN" \
              "https://api.github.com/repos/$REPO/releases/tags/$TAG")

            if [ "$RELEASE_EXISTS" != "200" ]; then
              # Buscar si hay algún release de esta app (cualquier versión)
              ANY_RELEASE=$(curl -s -H "Authorization: Bearer $GH_TOKEN" \
                "https://api.github.com/repos/$REPO/releases?per_page=100" \
                | jq -r --arg app "$APP_ID" '.[] | select(.tag_name | startswith($app + "_v")) | .tag_name' | head -1)

              if [ -z "$ANY_RELEASE" ]; then
                MESSAGES="${MESSAGES}⚠️ **${APP_ID}**: el info.md dice v${VERSION} pero **no tiene ningún release**\n"
                MESSAGES="${MESSAGES}   [Ver info.md](https://github.com/$REPO/blob/main/apps/${APP_ID}/info.md)\n"
              else
                MESSAGES="${MESSAGES}⚠️ **${APP_ID}**: el info.md dice v${VERSION} pero el último release es **${ANY_RELEASE}**\n"
                MESSAGES="${MESSAGES}   [Ver info.md](https://github.com/$REPO/blob/main/apps/${APP_ID}/info.md) | [Ver release](https://github.com/$REPO/releases/tag/${ANY_RELEASE})\n"
              fi
            fi
          done

          if [ -n "$MESSAGES" ]; then
            echo "Enviando alertas a Discord..."
            PAYLOAD=$(jq -n --arg content "🔍 **Revisión de versiones:**\n\n${MESSAGES}" '{content: $content}')
            curl -s -X POST -H "Content-Type: application/json" -d "$PAYLOAD" "$DISCORD_WEBHOOK"
            echo "Alertas enviadas."
          else
            echo "Todo en orden. Sin desajustes de versión."
          fi

// ================================================================
// 1. CONFIGURACIÓN

const CATALOG_URL = "catalog.json";
const GIST_DOWNLOADS_URL = "https://gist.githubusercontent.com/luisenriquepupo16-rgb/40391b1420b19d763eff10e35d1aec63/raw/downloads.json";

// ================================================================
// 2. UTILIDADES

/**
 * Formatea bytes a MB o KB.
 */
function formatSize(bytes) {
    if (!bytes) return "—";
    const mb = bytes / (1024 * 1024);
    return mb >= 1 ? `${mb.toFixed(2)} MB` : `${(bytes / 1024).toFixed(0)} KB`;
}

/**
 * Escapa HTML para prevenir XSS.
 */
function escapeHtml(str) {
    if (str === null || str === undefined) return "";
    return String(str)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
}

/**
 * Formatea un número con separador de miles.
 */
function formatNumber(n) {
    if (n === null || n === undefined) return "0";
    return n.toLocaleString("es-ES");
}

// ================================================================
// 3. TEMA CLARO / OSCURO
// ================================================================
function initTheme() {
    const toggle = document.getElementById("themeToggle");
    const icon = toggle.querySelector(".theme-icon");
    const label = toggle.querySelector(".theme-label");
    const saved = localStorage.getItem("theme") || "light";

    function applyTheme(theme) {
        if (theme === "dark") {
            document.documentElement.setAttribute("data-theme", "dark");
            icon.textContent = "☀️";
            label.textContent = "Modo claro";
        } else {
            document.documentElement.removeAttribute("data-theme");
            icon.textContent = "🌙";
            label.textContent = "Modo oscuro";
        }
    }

    applyTheme(saved);

    toggle.addEventListener("click", () => {
        const isDark = document.documentElement.getAttribute("data-theme") === "dark";
        const newTheme = isDark ? "light" : "dark";
        localStorage.setItem("theme", newTheme);
        applyTheme(newTheme);
    });
}

// ================================================================
// 4. CARGA DEL CATÁLOGO
// ================================================================
async function loadCatalog() {
    try {
        const response = await fetch(CATALOG_URL, { cache: "no-store" });
        if (!response.ok) throw new Error(`HTTP ${response.status}`);
        return await response.json();
    } catch (err) {
        console.error("Error cargando catalog.json:", err);
        return null;
    }
}

// ================================================================
// 5. CARGA DEL GIST DE DESCARGAS
// ================================================================
async function loadDownloads() {
    try {
        const response = await fetch(GIST_DOWNLOADS_URL, { cache: "no-store" });
        if (!response.ok) throw new Error(`HTTP ${response.status}`);
        const data = await response.json();
        return data.current || {};
    } catch (err) {
        console.warn("No se pudieron cargar las descargas:", err);
        return {};
    }
}

// ================================================================
// 6. RENDERIZADO DE PROYECTOS
// ================================================================
function renderProjects(apps, downloads) {
    const grid = document.getElementById("projectsGrid");

    if (!apps || apps.length === 0) {
        grid.innerHTML = `<p class="error-message">No hay proyectos publicados todavía.</p>`;
        return;
    }

    // Ordenar por fecha descendente
    const sorted = [...apps].sort((a, b) => {
        return (b.updated_at || "").localeCompare(a.updated_at || "");
    });

    grid.innerHTML = sorted.map(app => {
        const downloadsCount = downloads[app.id] || 0;
        const hasDownloads = downloadsCount > 0;

        const categoryHtml = app.category
            ? `<span class="category-tag">${escapeHtml(app.category)}</span>`
            : "";

        const dedicationHtml = app.dedication
            ? `<div class="dedication">💙 ${escapeHtml(app.dedication)}</div>`
            : "";

        const downloadsHtml = hasDownloads
            ? `<span class="downloads-badge">📥 ${formatNumber(downloadsCount)}</span>`
            : "";

        return `
            <article class="project-card" role="listitem">
                ${categoryHtml}
                <h3>${escapeHtml(app.name)}</h3>
                <p class="description">${escapeHtml(app.description)}</p>
                ${dedicationHtml}
                <div class="meta">
                    <span>📦 ${formatSize(app.apk?.size_bytes)}</span>
                    <span>📅 ${escapeHtml(app.updated_at)}</span>
                    ${downloadsHtml}
                </div>
                <a href="${escapeHtml(app.apk?.url || '#')}" 
                   target="_blank" 
                   rel="noopener" 
                   class="btn-primary"
                   style="text-align:center; padding:10px; font-size:0.9rem; margin-top:8px;">
                    📥 Descargar
                </a>
            </article>
        `;
    }).join("");
}

// ================================================================
// 7. INICIALIZACIÓN
// ================================================================
async function init() {
    initTheme();

    document.getElementById("year").textContent = new Date().getFullYear();

    // Cargar catálogo y descargas en paralelo
    const [catalog, downloads] = await Promise.all([
        loadCatalog(),
        loadDownloads()
    ]);

    if (!catalog) {
        document.getElementById("projectsGrid").innerHTML =
            `<p class="error-message">⚠️ No se pudieron cargar los proyectos. Intenta recargar la página.</p>`;
        return;
    }

    renderProjects(catalog.apps || [], downloads);
}

    <header class="article-header">
        <p class="article-meta">Octubre 2026 · Automatización</p>
        <h1>Cómo estructuré el catálogo de apps con un JSON generado automáticamente</h1>
    </header>

    <div class="article-content">

        <p>Una de las decisiones que más tiempo me ahorró en Träume des Blauen fue automatizar la generación del catálogo. En vez de mantener a mano una lista de aplicaciones, el catálogo se genera solo a partir de la estructura de carpetas. Este es el proceso.</p>

        <h2>El problema</h2>

        <p>Cuando empecé, tenía varias apps publicadas. Cada vez que añadía una nueva, tenía que:</p>

        <ul>
            <li>Actualizar la web con los datos de la app.</li>
            <li>Actualizar la app Android con los mismos datos.</li>
            <li>Mantener todo sincronizado.</li>
        </ul>

        <p>Era tedioso y propenso a errores. Se me olvidaba actualizar un lado u otro, y los datos quedaban inconsistentes.</p>

        <h2>La idea</h2>

        <p>La solución fue simple: <strong>separar los datos de la presentación</strong>. Los datos viven en un archivo JSON generado automáticamente. La presentación (web, app Android) solo lee ese JSON. Así, solo hay que mantener los datos en un sitio, y todo lo demás se actualiza solo.</p>

        <h2>La estructura de carpetas</h2>

        <p>Cada app vive en su propia carpeta dentro de <code>apps/</code>:</p>

        <pre><code>apps/
├── converty/
│   ├── converty_v1.2.0.apk
│   ├── icon.png
│   └── info.md
├── magic_world/
│   ├── magic_world_v2.1.10.apk
│   ├── icon.png
│   └── info.md
└── ...</code></pre>

        <p>El archivo <code>info.md</code> contiene los metadatos de la app en formato YAML (frontmatter) + cuerpo markdown. Por ejemplo:</p>

        <pre><code>---
name: Converty
version: 1.2.0
version_code: 5
package: com.converty.neki
category: Utilidades
technologies:
  - Kotlin
  - Apache POI
---

## Description

Aplicación Android que convierte documentos...</code></pre>

        <h2>El workflow</h2>

        <p>En GitHub, un workflow detecta cambios en la carpeta <code>apps/</code> y hace dos cosas:</p>

        <ol>
            <li>Crea un release de GitHub con el APK y el icono.</li>
            <li>Ejecuta un script que recorre todas las apps, lee su <code>info.md</code>, y genera el <code>catalog.json</code>.</li>
        </ol>

        <p>El script está escrito en PowerShell. Por cada app:</p>

        <ul>
            <li>Lee el frontmatter YAML.</li>
            <li>Busca el APK y el icono.</li>
            <li>Calcula el hash SHA-256 del APK.</li>
            <li>Construye una entrada del JSON con URLs apuntando al release.</li>
        </ul>

        <h2>El JSON resultante</h2>

        <p>El <code>catalog.json</code> tiene esta estructura:</p>

        <pre><code>{
  "version": 1,
  "generated_at": "2026-10-11T...",
  "site": {
    "title": "Traeume des Blauen",
    "author": "Dreamer"
  },
  "apps": [
    {
      "id": "converty",
      "name": "Converty",
      "version": "1.2.0",
      "category": "Utilidades",
      "icon": "https://...",
      "apk": {
        "url": "https://...",
        "size_bytes": 19626720,
        "sha256": "..."
      }
    }
  ]
}</code></pre>

        <p>Cada app tiene toda la información necesaria para ser mostrada y descargada.</p>

        <h2>Los consumidores</h2>

        <p>Dos cosas leen ese JSON:</p>

        <ul>
            <li><strong>La web del portafolio</strong>, que lo lee desde JavaScript y renderiza las tarjetas.</li>
            <li><strong>La app Android</strong>, que lo descarga con Ktor y lo cachea con DataStore.</li>
        </ul>

        <p>Ambos leen la misma URL: <code>https://luisenriquepupo16-rgb.github.io/Traeume-des-Blauen/catalog.json</code>. Cuando el catálogo cambia, ambos se actualizan solos.</p>

        <h2>Los detalles importantes</h2>

        <h3>Encabezados sin tildes</h3>

        <p>Un detalle que me costó descubrir: los encabezados del cuerpo del <code>info.md</code> (<code>## Description</code>, <code>## Uso</code>) tienen que estar en inglés y sin tildes. La razón es que PowerShell 5.1 puede corromper las tildes al leer el archivo, y el parser deja de encontrar los encabezados. El contenido sí puede estar en español.</p>

        <h3>Nombres de archivos sin caracteres especiales</h3>

        <p>Los APKs deben nombrarse como <code>&lt;app-id&gt;_v&lt;version&gt;.apk</code>, sin diéresis, sin mayúsculas. Los iconos siempre <code>icon.png</code>. Esto evita problemas de encoding en distintos sistemas.</p>

        <h3>El workflow hace commit del JSON</h3>

        <p>Después de generar el <code>catalog.json</code>, el workflow hace commit automático con un mensaje que incluye <code>[skip ci]</code>. Así no se dispara a sí mismo en bucle.</p>

        <h2>Lo que aprendí</h2>

        <p>Tres lecciones:</p>

        <ol>
            <li><strong>Separar datos de presentación es clave.</strong> Cuando los datos viven en un sitio y la presentación en otro, todo se actualiza solo. No hay que sincronizar nada a mano.</li>
            <li><strong>La automatización paga a largo plazo.</strong> Invertir 2 horas en configurar un workflow ahorra decenas de horas de mantenimiento manual.</li>
            <li><strong>Los detalles técnicos importan.</strong> El encoding, los nombres de archivos, los encabezados sin tildes... Son detalles pequeños que, si se ignoran, rompen todo el sistema.</li>
        </ol>

        <h2>Conclusión</h2>

        <p>Automatizar el catálogo fue una de las mejores decisiones del proyecto. Ahora añadir una app nueva es tan simple como crear una carpeta con tres archivos y hacer push. El resto se hace solo.</p>

        <p>Si estás manteniendo un catálogo a mano, mi recomendación es: automatízalo. No tiene que ser perfecto, solo tiene que quitar el trabajo manual del medio.</p>

    </div>

</article>

<article class="article">

    <header class="article-header">
        <p class="article-meta">Octubre 2026 · Desarrollo Android</p>
        <h1>Cómo resolví el problema de las descargas en background en Android</h1>
    </header>

    <div class="article-content">

        <p>Cuando empecé a desarrollar Träume des Blauen, la app que uso para publicar mis aplicaciones, no anticipé que las descargas serían uno de los problemas más complejos del proyecto. Este es el relato de cómo lo resolví, con contexto y sin edulcorar.</p>

        <h2>El problema</h2>

        <p>La app permite a los usuarios descargar APKs directamente. Al principio, la descarga se hacía en una corrutina lanzada desde el <code>ViewModel</code>. Funcionaba, pero tenía un fallo crítico: <strong>si el usuario salía de la app durante la descarga, la descarga se interrumpía</strong>.</p>

        <p>En Android, cuando una app pasa a segundo plano, el sistema puede matar el proceso si necesita recursos. La corrutina vivía en el <code>ViewModel</code>, que está atado al ciclo de vida de la Activity. Si la Activity moría, el ViewModel moría, y la corrutina con él.</p>

        <p>La consecuencia práctica: el usuario tenía que quedarse en la app durante toda la descarga. Si salía a hacer otra cosa, perdía el progreso. Aceptable para un APK pequeño, molesto para uno grande.</p>

        <h2>La primera solución (fallida): DownloadManager</h2>

        <p>Investigué y encontré el <code>DownloadManager</code> de Android. Es un servicio del sistema diseñado exactamente para esto: gestiona descargas en background, con notificación y todo. Parecía la solución perfecta.</p>

        <p>El problema: <strong>el <code>DownloadManager</code> no puede escribir en la carpeta privada de la app</strong> (<code>filesDir</code>). Es una carpeta aislada por seguridad, y el <code>DownloadManager</code> corre en un proceso distinto. Solo puede escribir en la carpeta de descargas pública.</p>

        <p>Probé varias alternativas, pero ninguna funcionaba limpiamente. Y necesitaba la carpeta privada porque la app instala los APKs con <code>FileProvider</code>, que requiere una ruta controlada.</p>

        <h2>La solución real: Foreground Service</h2>

        <p>La solución fue un <code>Foreground Service</code>. Es un servicio de Android que se ejecuta en primer plano, con una notificación persistente que informa al usuario de que algo está corriendo. El sistema garantiza que este servicio no se mata, incluso si la app pasa a background.</p>

        <p>La arquitectura quedó así:</p>

        <ul>
            <li>Un servicio llamado <code>DownloadService</code> que gestiona la descarga.</li>
            <li>Una corrutina dentro del servicio que descarga con Ktor.</li>
            <li>El servicio actualiza una notificación persistente con el progreso.</li>
            <li>La UI observa un <code>StateFlow</code> global (el <code>DownloadManagerSingleton</code>) para mostrar el progreso en pantalla.</li>
            <li>Al terminar, el servicio se detiene a sí mismo.</li>
        </ul>

        <p>Con esto, la descarga sobrevive a que la app pase a background. Si el usuario sale de la app, la descarga continúa. Al volver, ve el progreso actualizado.</p>

        <h2>Los detalles que aprendí por el camino</h2>

        <p>Implementar esto bien tiene varias trampas:</p>

        <h3>El tipo de Foreground Service</h3>

        <p>Desde Android 14, todos los <code>Foreground Service</code> deben declarar su tipo. Para descargas, el tipo es <code>dataSync</code>. Sin eso, el servicio no arranca.</p>

        <h3>El permiso de notificaciones</h3>

        <p>En Android 13+, hay que pedir el permiso <code>POST_NOTIFICATIONS</code> explícitamente. Si el usuario lo deniega, el servicio sigue funcionando, pero la notificación no se ve.</p>

        <h3>La cancelación real</h3>

        <p>Un bug que descubrí tarde: el botón "Cancelar" del diálogo cerraba el diálogo, pero no cancelaba la descarga. La descarga seguía en background y terminaba de todos modos.</p>

        <p>La solución fue que el botón "Cancelar" envíe una señal al servicio (<code>ACTION_CANCEL</code>) mediante un <code>Intent</code>. El servicio cancela el <code>Job</code> de la corrutina, borra el archivo parcial y se detiene.</p>

        <h3>El estado en el ViewModel</h3>

        <p>Antes de rediseñar, el estado del botón (<em>Descargar</em>, <em>Instalar</em>, <em>Actualizar</em>, <em>Abrir</em>) se recalculaba en cada recomposición con <code>remember</code>. Eso causaba bugs de timing: el botón no se actualizaba al instante.</p>

        <p>La solución fue centralizar el estado en un <code>Map&lt;String, AppStatus&gt;</code> en el ViewModel. La UI lee directamente de ahí. Cuando un paquete cambia (instalación, actualización, desinstalación), un <code>PackageChangeReceiver</code> dispara un recálculo.</p>

        <h2>Lo que aprendí</h2>

        <p>Tres lecciones que me llevo:</p>

        <ol>
            <li><strong>Las tareas largas no deben depender del ciclo de vida de la UI.</strong> Deben vivir en un servicio independiente. La UI es solo una vista del estado, no el estado mismo.</li>
            <li><strong>No todas las herramientas del sistema son universales.</strong> El <code>DownloadManager</code> es útil, pero no para este caso. A veces hay que bajar un nivel y hacer las cosas uno mismo.</li>
            <li><strong>Cuando algo falla por timing, la solución no es añadir más <code>delay()</code>.</strong> Es rediseñar la arquitectura para que no dependa del timing.</li>
        </ol>

        <h2>Conclusión</h2>

        <p>Este problema me llevó varios intentos y varias versiones. La solución final no es la más elegante del mundo, pero funciona, y sobrevive a los casos raros. Para una app personal como la mía, es suficiente.</p>

        <p>Si estás lidiando con algo similar, mi recomendación es: empieza con la solución simple (corrutina en el ViewModel), mide los problemas reales, y solo migra a <code>Foreground Service</code> si de verdad lo necesitas. La migración no es trivial.</p>

    </div>

        <p class="article-meta">Octubre 2026 · Arquitectura web</p>
        <h1>Cómo convertí la web del catálogo en un portafolio sin romper la app</h1>
    </header>

    <div class="article-content">

        <p>Este es un relato corto sobre una decisión arquitectónica que me permitió cambiar la cara pública de mi proyecto sin tocar la app que la consumía. La idea central: <strong>separar los datos de la presentación</strong> tiene ventajas que no siempre son obvias hasta que las necesitas.</p>

        <h2>El contexto</h2>

        <p>Träume des Blauen tiene dos componentes:</p>

        <ul>
            <li><strong>Una web publicada en GitHub Pages</strong>, que era originalmente un catálogo de apps.</li>
            <li><strong>Una app Android</strong>, que lee el mismo catálogo para mostrar las apps disponibles.</li>
        </ul>

        <p>Ambos leían el mismo archivo: <code>docs/catalog.json</code>, publicado en <code>https://luisenriquepupo16-rgb.github.io/Traeume-des-Blauen/catalog.json</code>.</p>

        <p>El problema: la web ya no la usaba nadie. La app es la interfaz real. La web era un remanente de la primera versión, cuando el catálogo era solo web. Pero seguía publicada, y seguía consumiendo recursos (aunque mínimos).</p>

        <h2>La idea</h2>

        <p>Quería convertir la web en un <strong>portafolio personal</strong>. Un lugar donde mostrar mis proyectos, mis retos técnicos, mi blog, y mi contacto. Algo que me representara, no solo una lista de apps.</p>

        <p>Pero había un problema: si tocaba la web, podía romper la app. La app depende de <code>catalog.json</code>. Si ese archivo desaparecía o cambiaba de URL, la app dejaba de funcionar.</p>

        <p>La solución fue simple: <strong>mantener el JSON intacto, y reescribir solo la presentación</strong>.</p>

        <h2>Cómo lo hice</h2>

        <p>Tres reglas:</p>

        <ol>
            <li><strong>No tocar <code>catalog.json</code>.</strong> El archivo sigue en la misma URL, con la misma estructura. La app lo sigue leyendo igual.</li>
            <li><strong>No tocar el workflow</strong> que genera el JSON. Sigue funcionando igual.</li>
            <li><strong>Reescribir <code>index.html</code>, <code>style.css</code> y <code>app.js</code>.</strong> Estos archivos son solo presentación. La app no los usa.</li>
        </ol>

        <p>Con esas tres reglas, la web antigua desapareció y apareció el portafolio. La app siguió funcionando igual, porque nunca dependió de la web, solo del JSON.</p>

        <h2>Por qué funcionó</h2>

        <p>Esta separación entre datos y presentación es una lección clásica de arquitectura de software, pero a veces se olvida. Cuando los datos viven en un sitio y la presentación en otro, puedes cambiar la presentación sin tocar los datos.</p>

        <p>En este caso:</p>

        <ul>
            <li>Los datos: <code>catalog.json</code>.</li>
            <li>Presentación web: <code>index.html</code>, <code>style.css</code>, <code>app.js</code>.</li>
            <li>Presentación Android: la app nativa.</li>
        </ul>

        <p>Los datos son la fuente única de verdad. Ambas presentaciones los leen. Cambiar una presentación no afecta a la otra.</p>

        <h2>El detalle que casi se me escapa</h2>

        <p>Cuando estaba planeando la conversión, casi cometo un error: pensé en renombrar <code>docs/</code> a algo más apropiado, como <code>web/</code>. Pero GitHub Pages sirve desde <code>docs/</code> por defecto. Si lo renombraba, la URL del <code>catalog.json</code> cambiaba, y la app dejaba de encontrarlo.</p>

        <p>Así que mantuve <code>docs/</code>. Menos elegante, pero funciona.</p>

        <p>Otra cosa que aprendí: la URL de GitHub Pages es <code>usuario.github.io/repositorio/</code>, y sirve desde la carpeta configurada (<code>docs/</code> en este caso). Si algún día cambio la carpeta, tengo que actualizar la URL en la app también. Es un acoplamiento que hay que tener presente.</p>

        <h2>Lo que aprendí</h2>

        <p>Tres lecciones:</p>

        <ol>
            <li><strong>Separar datos de presentación es clave.</strong> No es solo una buena práctica teórica, tiene consecuencias prácticas cuando quieres cambiar una cosa sin tocar la otra.</li>
            <li><strong>Las URLs son contratos.</strong> Si cambias una URL, rompes a todos los que dependen de ella. En este caso, la app dependía de la URL del JSON. Cambiarla habría sido un error grave.</li>
            <li><strong>Los detalles de infraestructura importan.</strong> La carpeta <code>docs/</code>, el nombre del repositorio, la configuración de GitHub Pages... Son detalles que determinan la URL final. Hay que entenderlos antes de tocar nada.</li>
        </ol>

        <h2>Conclusión</h2>

        <p>Convertir la web en un portafolio fue una decisión que me permitió tener una presencia pública más coherente con lo que hago, sin arriesgar la app. La clave fue entender qué partes del sistema eran datos y cuáles eran presentación, y tratar cada una con las reglas correspondientes.</p>

        <p>Si estás pensando en rediseñar algo en tu proyecto, mi recomendación es: identifica primero qué es dato y qué es presentación. Cambia la presentación todo lo que quieras. Los datos, trátalos con cuidado.</p>

    </div>

                <h3>Transsion bloquea <code>Android/data/</code></h3>
                <p class="challenge-context">En dispositivos Transsion (marca común en Cuba), el sistema bloquea el acceso a <code>Android/data/</code>. Las apps que guardan datos ahí no pueden leerlos.</p>
                <p class="challenge-solution"><strong>Solución:</strong> usar <code>context.filesDir</code> en lugar de <code>Android/data/</code>. Es una carpeta privada de la app, siempre accesible.</p>
                <p class="challenge-lesson"><strong>Aprendizaje:</strong> nunca asumir que las APIs estándar funcionan igual en todas las marcas.</p>
            </article>

            <article class="challenge-card">
                <h3>DownloadManager no puede escribir en <code>filesDir</code></h3>
                <p class="challenge-context"><code>DownloadManager</code> es un servicio del sistema que gestiona descargas. Pero no puede escribir en la carpeta privada de la app (<code>filesDir</code>), porque es una carpeta aislada.</p>
                <p class="challenge-solution"><strong>Solución:</strong> sustituir <code>DownloadManager</code> por Ktor (cliente HTTP) y descargar directamente con <code>FileOutputStream</code>.</p>
                <p class="challenge-lesson"><strong>Aprendizaje:</strong> no todas las herramientas del sistema son universales. A veces hay que bajar un nivel y hacerlo uno mismo.</p>
            </article>

            <article class="challenge-card">
                <h3>El instalador no abre después de descargar</h3>
                <p class="challenge-context">Después de descargar un APK, el botón "Instalar" no se actualizaba correctamente. A veces se quedaba en "Instalar" después de instalar, a veces no cambiaba a "Abrir".</p>
                <p class="challenge-solution"><strong>Solución:</strong> rediseñar el sistema de estados. Usar un <code>Foreground Service</code> para descargas, un <code>PackageChangeReceiver</code> para detectar cambios en paquetes, y un <code>Map&lt;String, AppStatus&gt;</code> en el ViewModel como única fuente de verdad.</p>
                <p class="challenge-lesson"><strong>Aprendizaje:</strong> cuando algo falla por "timing", la solución no es añadir más <code>delay()</code>. Es rediseñar la arquitectura para que no dependa del timing.</p>
            </article>

            <article class="challenge-card">
                <h3>Perder el progreso de descarga al salir de la app</h3>
                <p class="challenge-context">Si el usuario salía de la app durante una descarga, la descarga se interrumpía. Al volver, había que empezar de cero.</p>
                <p class="challenge-solution"><strong>Solución:</strong> migrar de una corrutina en el ViewModel a un <code>Foreground Service</code> con notificación persistente. La descarga sobrevive a que la app pase a background.</p>
                <p class="challenge-lesson"><strong>Aprendizaje:</strong> las tareas largas no deben depender del ciclo de vida de la UI. Deben vivir en un servicio independiente.</p>
            </article>

            <article class="challenge-card">
                <h3>Cancelar una descarga no la cancelaba</h3>
                <p class="challenge-context">El botón "Cancelar" del diálogo cerraba el diálogo, pero la descarga seguía en segundo plano. Al terminar, el archivo aparecía completo.</p>
                <p class="challenge-solution"><strong>Solución:</strong> el botón "Cancelar" envía una señal al servicio (<code>ACTION_CANCEL</code>), que cancela el <code>Job</code> de la corrutina y borra el archivo parcial.</p>
                <p class="challenge-lesson"><strong>Aprendizaje:</strong> "Cancelar" en una UI no es solo cerrar el diálogo. Es detener el proceso real.</p>
            </article>

import os

# 1. LISTA NEGRA AUTOMÁTICA
FILTROS_SISTEMA = [
    ".git", "__pycache__", "node_modules", ".venv", "venv", 
    "env", ".vscode", ".idea", "migrations", "package-lock.json",
    "yarn.lock", "dist", "build", ".next", ".nuxt", "estructura.txt"
]

# 2. EXTENSIONES DE ARCHIVOS BINARIOS A IGNORAR
EXTENSIONES_BINARIAS = [
    ".png", ".jpg", ".jpeg", ".gif", ".bmp", ".webp", ".ico", ".svg",
    ".mp3", ".mp4", ".wav", ".ogg", ".flac",
    ".zip", ".rar", ".7z", ".tar", ".gz", ".apk", ".aab",
    ".pdf", ".doc", ".docx", ".xls", ".xlsx", ".ppt", ".pptx",
    ".exe", ".dll", ".so", ".dylib",
    ".keystore", ".jks", ".p12", ".pfx",
    ".ttf", ".otf", ".woff", ".woff2"
]

# 3. LISTA PERSONALIZABLE
carpetas_personalizadas = []
mostrar_archivos = True

def es_archivo_binario(nombre_archivo: str) -> bool:
    """Verifica si un archivo tiene extensión binaria."""
    nombre_lower = nombre_archivo.lower()
    for ext in EXTENSIONES_BINARIAS:
        if nombre_lower.endswith(ext):
            return True
    return False

def leer_contenido_archivo(ruta_archivo):
    """Intenta leer el contenido de un archivo de texto. Si es binario, retorna un mensaje."""
    # Verificar si es archivo binario por extensión
    if es_archivo_binario(ruta_archivo):
        return "[ARCHIVO BINARIO - CONTENIDO NO LEGIBLE]"
    
    try:
        with open(ruta_archivo, "r", encoding="utf-8", errors="ignore") as f:
            contenido = f.read()
            # Si el archivo tiene caracteres nulos o es muy corto (posible binario), marcarlo
            if "\x00" in contenido:
                return "[ARCHIVO BINARIO - CONTENIDO NO LEGIBLE]"
            return contenido
    except Exception as e:
        return f"[No se pudo leer: {e}]"

def generar_arbol(ruta_dir, prefijo="", ignorar=None, contenido_archivos=None):
    """
    Genera el árbol ignorando carpetas clave y recolecta contenido de archivos.
    """
    if ignorar is None:
        ignorar = []
    if contenido_archivos is None:
        contenido_archivos = []
    
    todos_los_filtros = set(FILTROS_SISTEMA + ignorar)
    
    try:
        contenido_total = sorted(os.listdir(ruta_dir))
        if not mostrar_archivos:
            contenido_total = [f for f in contenido_total if os.path.isdir(os.path.join(ruta_dir, f))]

        contenido = [f for f in contenido_total if f not in todos_los_filtros]
    except PermissionError:
        return [f"{prefijo}└── [Acceso Denegado]"], []

    lineas = []
    for i, elemento in enumerate(contenido):
        ruta_completa = os.path.join(ruta_dir, elemento)
        es_el_ultimo = (i == len(contenido) - 1)
        
        conector = "└── " if es_el_ultimo else "├── "
        lineas.append(f"{prefijo}{conector}{elemento}")
        
        if os.path.isdir(ruta_completa):
            nuevo_prefijo = prefijo + ("    " if es_el_ultimo else "│   ")
            sub_lineas, sub_contenido = generar_arbol(ruta_completa, nuevo_prefijo, ignorar)
            lineas.extend(sub_lineas)
            contenido_archivos.extend(sub_contenido)
        elif mostrar_archivos and os.path.isfile(ruta_completa):
            # Extraer contenido solo si no es binario
            texto = leer_contenido_archivo(ruta_completa)
            contenido_archivos.append(f"\n{'='*20}\nARCHIVO: {ruta_completa}\n{'='*20}\n{texto}")
            
    return lineas, contenido_archivos

def guardar_estructura():
    directorio_actual = os.getcwd()
    nombre_archivo = "estructura.txt"
    
    print(f"\n📂 Escaneando: {directorio_actual}...")
    
    arbol, contenido_files = generar_arbol(directorio_actual, ignorar=carpetas_personalizadas)
    
    resultado_final = [
        f"ESTRUCTURA DEL PROYECTO: {directorio_actual}",
        "=" * (len(directorio_actual) + 25),
        "",
        "."
    ] + arbol + ["\n\n" + "="*50 + "\nCONTENIDO DE LOS ARCHIVOS\n" + "="*50] + contenido_files
    
    try:
        with open(nombre_archivo, "w", encoding="utf-8") as f:
            f.write("\n".join(resultado_final))
        print(f"\n✅ Estructura y contenido guardados en '{nombre_archivo}'.")
    except Exception as e:
        print(f"\n❌ Error al guardar: {e}")

def menu_principal():
    global carpetas_personalizadas, mostrar_archivos
    while True:
        estado_archivos = "ON" if mostrar_archivos else "OFF"
        print(f"""

1. 🚀 Comenzar escaneo
2. ⚙️ Agregar filtros extra

4. 🧹 Limpiar filtros

        elif opcion == "3":
            mostrar_archivos = not mostrar_archivos
        elif opcion == "5":
            break

if __name__ == "__main__":
    menu_principal()

# --- Función auxiliar: leer un archivo como UTF-8 sin BOM ---
function Read-Utf8NoBom {
    param([string]$Path)
    $bytes = [System.IO.File]::ReadAllBytes($Path)
    return [System.Text.Encoding]::UTF8.GetString($bytes)
}

# --- Función auxiliar: parsear frontmatter de un info.md ---
function Get-Frontmatter {
    param([string]$Path)

    $content = Read-Utf8NoBom -Path $Path

    if ($content -notmatch '(?s)^\s*---\s*\r?\n(.*?)\r?\n---\s*\r?\n(.*)$') {
        throw "El archivo $Path no tiene frontmatter YAML valido."
    }

    $yamlText = $Matches[1]
    $bodyText = $Matches[2]

    $meta = ConvertFrom-Yaml $yamlText
    return @{
        Meta = $meta
        Body = $bodyText.Trim()
    }
}

# --- Función auxiliar: extraer una sección del markdown ---
function Get-MarkdownSection {
    param(
        [string]$Body,
        [string]$Heading
    )
    $pattern = "(?s)##\s+$([regex]::Escape($Heading))\s*\r?\n(.*?)(?=\r?\n##\s|\z)"
    if ($Body -match $pattern) {
        return $Matches[1].Trim()
    }
    return $null
}

def guardar_historial(data):
    try:
        conversations = data.get("conversations", {})

        if len(conversations) > MAX_CONVERSACIONES:
            sorted_convs = sorted(
                conversations.items(),
                key=lambda x: x[1].get("last_activity", ""),
                reverse=True
            )
            conversations = dict(sorted_convs[:MAX_CONVERSACIONES])

        cutoff = time.time() - (MAX_DIAS * 86400)
        for conv_id in list(conversations.keys()):
            last = conversations[conv_id].get("last_activity", "")
            try:
                last_ts = time.mktime(time.strptime(last, "%Y-%m-%dT%H:%M:%SZ"))
                if last_ts < cutoff:
                    del conversations[conv_id]
            except Exception:
                pass

        data["conversations"] = conversations

        requests.patch(
            f"https://api.github.com/gists/{GIST_ID}",
            headers={"Authorization": f"Bearer {GIST_TOKEN}"},
            json={
                "files": {
                    "chat_history.json": {
                        "content": json.dumps(data, ensure_ascii=False, indent=2)
                    }
                }
            },
            timeout=30
        )
        print("Historial guardado en Gist.")
    except Exception as e:
        print(f"Error guardando historial: {e}")

def llamar_gemini(prompt, max_reintentos=3):
    for intento in range(max_reintentos):
        try:
            response = requests.post(
                GEMINI_URL,
                headers={"Content-Type": "application/json"},
                json={
                    "contents": [{
                        "parts": [{"text": prompt}]
                    }]
                },
                timeout=60
            )

            data = response.json()

            if data.get("error", {}).get("code") == 503:
                espera = (intento + 1) * 5
                print(f"Modelo saturado. Reintentando en {espera}s...")
                time.sleep(espera)
                continue

            candidates = data.get("candidates", [])
            if candidates:
                parts = candidates[0].get("content", {}).get("parts", [])
                if parts:
                    return parts[0].get("text", "Sin respuesta")

            print(f"Error de Gemini: {data.get('error', {}).get('message', 'Desconocido')}")
            time.sleep(3)

        except Exception as e:
            print(f"Error en intento {intento + 1}: {e}")
            time.sleep(3)

    return "Lo siento, no puedo responder en este momento. Intenta de nuevo en unos minutos."

    relevantes = []
    for job in jobs:
        titulo = job.get("title", "").lower()
        descripcion = job.get("description", "").lower()

            es_relevante = False
            if any(k in texto for k in PALABRAS_FUERTES):
                es_relevante = True
            elif any(k in texto for k in PALABRAS_DEBILES) and \
                 any(r in texto for r in ["remoto", "remote"]):
                es_relevante = True

            if not es_relevante:
                continue

            h = hash_oferta(msg.text)
            if h in hashes_telegram:
                continue

            mensajes_relevantes.append((msg.text, h))

        if not mensajes_relevantes:
            print("Sin ofertas nuevas relevantes.")
            return

        cuerpo = "\n\n---\n\n".join(m[0] for m in mensajes_relevantes[:5])
        aviso = f"Ofertas relevantes en CubanTech Jobs:\n\n{cuerpo}"

        payload = {"content": aviso}
        requests.post(DISCORD_WEBHOOK, json=payload)
        print(f"Enviadas {len(mensajes_relevantes[:5])} ofertas nuevas a Discord.")

        # Guardar hashes nuevos
        for _, h in mensajes_relevantes[:5]:
            hashes_telegram.add(h)

        hashes["telegram"] = list(hashes_telegram)
        guardar_hashes(hashes)


asyncio.run(main())
